```c++
int removeElement(vector<int>& nums, int val) {
    int left = 0, right = nums.size();
    while (left < right) {
        if (nums[left] == val) {
            nums[left] = nums[right - 1];
            --right;
        } else {
            ++left;
        }
    }
    return left;
}
```

```c++
int removeDuplicates(vector<int> &nums) {
    int n = nums.size();
    if (n == 0)return 0;
    int fast = 1, slow = 1;
    while (fast < n) {
        if (nums[fast] != nums[fast - 1]) {
            nums[slow] = nums[fast];
            ++slow;// 满足条件慢指针动
        }
        ++fast;
    }
    return slow;
}
```

## 滑动窗口

[209. 长度最小的子数组](https://leetcode.cn/problems/minimum-size-subarray-sum/)

```c++
int minSubArrayLen(int target, vector<int> &nums) {
    int n = nums.size(); if (n == 0) return 0;
    int ans = INT_MAX, sum = 0;
    int left = 0, right = 0;
    while (right < n) {
        sum += nums[right];
        while (sum >= target) {
            ans = min(ans, right - left + 1);
            sum -= nums[left];
            left++;
        }
        right++;
    }
    return ans == INT_MAX ? 0 : ans;
}
```

```c++
// 1. 前序遍历：根 → 左 → 右
void preOrder(TreeNode* root, vector<int>& res) {
    if (!root) return; // 递归终止条件：节点为空
    res.push_back(root->val);  // 访问根节点
    preOrder(root->left, res); // 递归左子树
    preOrder(root->right, res);// 递归右子树
}

// 2. 中序遍历：左 → 根 → 右
void inOrder(TreeNode* root, vector<int>& res) {
    if (!root) return;
    inOrder(root->left, res);
    res.push_back(root->val);
    inOrder(root->right, res);
}

// 3. 后序遍历：左 → 右 → 根
void postOrder(TreeNode* root, vector<int>& res) {
    if (!root) return;
    postOrder(root->left, res);
    postOrder(root->right, res);
    res.push_back(root->val);
}
```

### 迭代

```c++
// 非递归前序遍历
void preOrderNonRecur(TreeNode* root, vector<int>& res) {
    if (!root) return;
    stack<TreeNode*> st;
    st.push(root);
    // 栈：先进后出，所以先压右孩子，再压左孩子
    while (!st.empty()) {
        TreeNode* node = st.top();
        st.pop();
        res.push_back(node->val);
        if (node->right) st.push(node->right);
        if (node->left) st.push(node->left);
    }
}

// 非递归中序遍历
void inOrderNonRecur(TreeNode* root, vector<int>& res) {
    if (!root) return;
    stack<TreeNode*> st;
    TreeNode* cur = root;
    while (cur || !st.empty()) {
        // 一直向左走，压入所有左节点
        while (cur) {
            st.push(cur);
            cur = cur->left;
        }
        // 取出节点访问
        cur = st.top();
        st.pop();
        res.push_back(cur->val);
        // 转向右子树
        cur = cur->right;
    }
}

// 非递归后序遍历
void postOrderNonRecur(TreeNode* root, vector<int>& res) {
    if (!root) return;
    stack<TreeNode*> st;
    TreeNode* cur = root;
    TreeNode* lastVisited = nullptr; // 记录上一个访问的节点
    
    while (cur || !st.empty()) {
        // 一直向左
        while (cur) {
            st.push(cur);
            cur = cur->left;
        }
        
        TreeNode* topNode = st.top();
        // 右子树为空 或 右子树已访问 → 访问当前节点
        if (!topNode->right || topNode->right == lastVisited) {
            res.push_back(topNode->val);
            lastVisited = topNode;
            st.pop();
        } else {
            // 否则转向右子树
            cur = topNode->right;
        }
    }
}
```

## BFS

```c++
// 1. 基础 BFS：一维数组（按层遍历）
void bfs(TreeNode* root, vector<int>& res) {
    if (!root) return;
    
    queue<TreeNode*> q;  // 队列：先进先出
    q.push(root);        // 根节点入队
    
    while (!q.empty()) {
        TreeNode* cur = q.front(); // 取队首
        q.pop();                   // 出队
        
        res.push_back(cur->val);   // 访问节点
        
        if (cur->left)  q.push(cur->left);  // 左孩子入队
        if (cur->right) q.push(cur->right); // 右孩子入队
    }
}

// 2. 分层 BFS：二维数组（每层单独一组）
vector<vector<int>> bfsLevel(TreeNode* root) {
    vector<vector<int>> res;
    if (!root) return res;
    
    queue<TreeNode*> q;
    q.push(root);
    
    while (!q.empty()) {
        int levelSize = q.size(); // 当前层有多少个节点
        vector<int> level;        // 存储当前层的结果
        
        // 遍历当前层所有节点
        for (int i = 0; i < levelSize; ++i) {
            TreeNode* cur = q.front();
            q.pop();
            
            level.push_back(cur->val);
            
            if (cur->left)  q.push(cur->left);
            if (cur->right) q.push(cur->right);
        }
        
        res.push_back(level); // 把当前层加入结果
    }
    return res;
}
```

## DFS

### 递归

```c++
void dfs(int u, vector<vector<int>>& adj, vector<bool>& visited, vector<int>& res) {
    visited[u] = true;   // 标记当前节点已访问
    res.push_back(u);     // 记录访问顺序

    // 遍历所有相邻节点
    for (int v : adj[u]) {
        if (!visited[v]) { // 未访问过才递归
            dfs(v, adj, visited, res);
        }
    }
}
```

### 迭代

```c++
void dfsNonRecur(int start, vector<vector<int>>& adj, vector<bool>& visited, vector<int>& res) {
    stack<int> st;
    st.push(start);
    visited[start] = true;

    while (!st.empty()) {
        int u = st.top();
        st.pop();
        res.push_back(u);

        // 逆序入栈，保证顺序和递归一致（可选）
        for (auto it = adj[u].rbegin(); it != adj[u].rend(); ++it) {
            int v = *it;
            if (!visited[v]) {
                visited[v] = true;
                st.push(v);
            }
        }
    }
}
```

[200. 岛屿数量](https://leetcode.cn/problems/number-of-islands/)

```c++
class Solution {
public:
    int numIslands(vector<vector<char>>& grid) {
        // 四个方向
        
        int cnt = 0;
        int m = grid.size();
        if (m == 0) return 0;
        int n = grid[0].size();

        // 遍历每个格子
        for (int i = 0; i < m; ++i) {
            for (int j = 0; j < n; ++j) {
                if (grid[i][j] == '1') {
                    cnt++;
                    // 调用独立的 DFS 函数
                    dfs(grid, direction, m, n, i, j);
                }
            }
        }
        return cnt;
    }

private:
    void dfs(vector<vector<char>>& grid, 
             int m, int n, 
             int x, int y) {
        vector<pair<int, int>> direction{{0, 1}, {1, 0}, {-1, 0}, {0, -1}};
        stack<pair<int, int>> st;
        st.push({x, y});
        grid[x][y] = '0';  // 标记已访问

        while (!st.empty()) {
            auto [curX, curY] = st.top();
            st.pop();

            // 遍历四个方向
            for (auto& dir : direction) {
                int dx = dir.first;
                int dy = dir.second;
                int nx = curX + dx;
                int ny = curY + dy;

                if (nx >= 0 && nx < m && ny >= 0 && ny < n && grid[nx][ny] == '1') {
                    grid[nx][ny] = '0';
                    st.push({nx, ny});
                }
            }
        }
    }
};
```

[1926. 迷宫中离入口最近的出口](https://leetcode.cn/problems/nearest-exit-from-entrance-in-maze/)

```c++
int nearestExit(vector<vector<char>>& maze, vector<int>& entrance) {
    int m = maze.size();
    int n = maze[0].size();
    const static vector<pair<int, int>> DIRS = {{1, 0}, {0, 1}, {-1, 0}, {0, -1}};

    auto bfs = [&](int x, int y) -> int {
        queue<pair<int, int>> q;
        vector<vector<bool>> visited(m, vector<bool>(n, false));
        q.push({x, y});
        visited[x][y] = true;

        int step = 0;
        while (!q.empty()) {
            int sz = q.size();
            for (int i = 0; i < sz; i++) {
                auto [curX, curY] = q.front();
                q.pop();

                for (auto [dx, dy] : DIRS) {
                    int nx = curX + dx;
                    int ny = curY + dy;

                    if (nx < 0 || nx >= m || ny < 0 || ny >= n)
                        continue;
                    if (maze[nx][ny] != '.' || visited[nx][ny])
                        continue;

                    if (nx == 0 || nx == m - 1 || ny == 0 || ny == n - 1) {
                        return step + 1;
                    }

                    visited[nx][ny] = true;
                    q.push({nx, ny});
                }
            }
            step++;
        }
        return -1;
    };

    return bfs(entrance[0], entrance[1]);
}
```
