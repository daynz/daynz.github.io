### 1. 冒泡排序
#### 思路
重复遍历数组，依次比较相邻元素，若顺序错误则交换；每一轮遍历会将当前未排序部分的最大元素“冒泡”到末尾；加入标志位优化，若某轮无交换则说明数组已有序，提前终止，减少不必要的遍历。

#### 代码
```cpp
void bubbleSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n; ++i) {
        bool swapped = false; // 标记本轮是否有交换
        // 最后i个元素已排序，无需比较
        for (int j = 0; j < n - i - 1; ++j) {
            if (arr[j] > arr[j + 1]) {
                swap(arr[j], arr[j + 1]); // 交换相邻元素
                swapped = true;
            }
        }
        if (!swapped) break; // 无交换则提前退出
    }
}
```
#### 重点
- 时间复杂度：平均/最坏 $O(n^2)$，最好（近乎有序）$O(n)$；
- 空间复杂度 $O(1)$，**稳定排序**、原地排序；
- 核心优化点是“交换标志位”，避免对已排序数组的无效遍历；
- 仅适用于小规模或几乎有序的数据排序。

### 2. 选择排序
#### 思路
将数组分为“已排序区”和“未排序区”，每轮从未排序区找到最小（大）元素，与未排序区的第一个元素交换；逐步扩大已排序区，直到整个数组有序。

#### 代码
```cpp
void selectionSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 0; i < n - 1; ++i) {
        int minIdx = i; // 记录未排序区最小元素的索引
        // 遍历未排序区，找最小元素
        for (int j = i + 1; j < n; ++j) {
            if (arr[j] < arr[minIdx]) {
                minIdx = j;
            }
        }
        // 交换最小元素与未排序区第一个元素
        swap(arr[i], arr[minIdx]);
    }
}
```
#### 重点
- 时间复杂度：平均/最坏/最好均为 $O(n^2)$，无最优情况；
- 空间复杂度 $O(1)$，**不稳定排序**、原地排序；
- 交换次数少（最多 $n-1$ 次），适合交换成本高的场景；

### 3. 插入排序
#### 思路
将数组分为“已排序区”和“未排序区”，依次取未排序区的元素，插入到已排序区的合适位置（从后往前比较，大于当前元素则后移）；直到所有元素插入完成。

#### 代码
```cpp
void insertionSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 1; i < n; ++i) {
        int key = arr[i]; // 待插入的元素
        int j = i - 1;
        // 已排序区元素大于key时，向后移动
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            --j;
        }
        arr[j + 1] = key; // 插入到正确位置
    }
}
```
#### 重点
- 时间复杂度：平均/最坏 $O(n^2)$，最好（近乎有序）$O(n)$；
- 空间复杂度 $O(1)$，**稳定排序**、原地排序；
- 对近乎有序的小规模数据效率极高，是希尔排序的基础；
- 插入过程仅移动元素，无频繁交换，实际执行效率优于冒泡/选择排序。

### 4. 希尔排序
#### 思路
插入排序的改进版，核心是“分组插入排序”：先将数组按步长 `gap` 分组，对每组进行插入排序；逐步缩小步长（通常减半），直到步长为1，此时数组已基本有序，最后做一次普通插入排序。

#### 代码
```cpp
void shellSort(vector<int>& arr) {
    int n = arr.size();
    // 初始步长为n/2，逐步减半
    for (int gap = n / 2; gap > 0; gap /= 2) {
        // 对每组进行插入排序
        for (int i = gap; i < n; ++i) {
            int key = arr[i];
            int j = i;
            // 组内元素比较，大于key则后移（步长gap）
            while (j >= gap && arr[j - gap] > key) {
                arr[j] = arr[j - gap];
                j -= gap;
            }
            arr[j] = key;
        }
    }
}
```
#### 重点
- 时间复杂度：平均 $O(n^{1.3})$，最坏 $O(n^2)$，最好 $O(n)$；
- 空间复杂度 $O(1)$，**不稳定排序**、原地排序；
- 步长选择影响效率，常用步长序列：$n/2, n/4,...,1$；
- 相比简单排序，对中等规模数据效率提升显著。

### 5. 快速排序
#### 思路
分治思想：选一个“基准值”，将数组分为“小于基准”和“大于基准”两部分，基准归位；递归对左右子数组重复上述操作，直到子数组长度为1（天然有序）。优化：选基准时用“三数取中法”（避免最坏情况），小规模子数组改用插入排序。

#### 代码
```cpp
int partition(vector<int>& arr, int low, int high) {
    // 三数取中法选基准（优化，避免最坏情况）
    int mid = low + (high - low) / 2;
    if (arr[mid] > arr[high]) swap(arr[mid], arr[high]);
    if (arr[low] > arr[high]) swap(arr[low], arr[high]);
    if (arr[mid] > arr[low]) swap(arr[mid], arr[low]);
    int pivot = arr[low]; // 基准值
    
    int i = low, j = high;
    while (i < j) {
        // 从右找小于基准的元素
        while (i < j && arr[j] >= pivot) --j;
        arr[i] = arr[j];
        // 从左找大于基准的元素
        while (i < j && arr[i] <= pivot) ++i;
        arr[j] = arr[i];
    }
    arr[i] = pivot; // 基准归位
    return i;
}

// 快速排序递归函数
void quickSortHelper(vector<int>& arr, int low, int high) {
    // 小规模子数组改用插入排序（优化）
    if (high - low + 1 <= 10) {
        for (int i = low + 1; i <= high; ++i) {
            int key = arr[i];
            int j = i - 1;
            while (j >= low && arr[j] > key) {
                arr[j + 1] = arr[j];
                --j;
            }
            arr[j + 1] = key;
        }
        return;
    }
    if (low < high) {
        int pivotIdx = partition(arr, low, high);
        quickSortHelper(arr, low, pivotIdx - 1); // 左子数组
        quickSortHelper(arr, pivotIdx + 1, high); // 右子数组
    }
}

void quickSortIterative(vector<int>& arr) {
    int n = arr.size();
    if (n <= 1) return;

    // 创建一个栈来存储待处理的区间 [low, high]
    stack<pair<int, int>> s;

    s.push({0, n - 1});

    while (!s.empty()) {
        // 弹出一个区间
        pair<int, int> current = s.top();
        s.pop();
        int low = current.first;
        int high = current.second;

        // 如果区间内只有一个或没有元素，无需处理
        if (low >= high) continue;

        // 进行分区操作，获取基准点的索引
        int pi = partition(arr, low, high);

        // 【优化策略】：先压入较大的区间，再压入较小的区间
        // 这样可以保证栈的最大深度为 O(log n)，避免退化为 O(n)
        // 左子区间: [low, pi - 1]
        // 右子区间: [pi + 1, high]

        if (pi - 1 > low) {
            s.push({low, pi - 1});
        }
        if (pi + 1 < high) {
            s.push({pi + 1, high});
        }
    }
}

// 对外接口
void quickSort(vector<int>& arr) {
    if (arr.empty()) return;
    quickSortHelper(arr, 0, arr.size() - 1);
}
```
#### 重点
- 时间复杂度：平均/最好 $O(n\log n)$，最坏（已排序数组，未优化）$O(n^2)$；
- 空间复杂度 $O(\log n)$（递归栈），**不稳定排序**、原地排序；
- 工程中首选的通用排序算法，优化后可避免最坏情况；
- 分区操作是核心，基准归位后左右子数组独立排序。

### 6. 归并排序
#### 思路
分治思想：将数组递归拆分为两个子数组，直到子数组长度为1；然后将两个有序子数组合并为一个有序数组，逐层合并直到整个数组有序。合并过程需要额外空间存储临时数据。

#### 代码
```cpp
// 合并两个有序子数组：arr[low..mid] 和 arr[mid+1..high]
void merge(vector<int>& arr, int low, int mid, int high) {
    vector<int> temp(high - low + 1); // 临时数组
    int i = low, j = mid + 1, k = 0;
    
    // 合并两个有序子数组到临时数组
    while (i <= mid && j <= high) {
        if (arr[i] <= arr[j]) {
            temp[k++] = arr[i++];
        } else {
            temp[k++] = arr[j++];
        }
    }
    // 处理剩余元素
    while (i <= mid) temp[k++] = arr[i++];
    while (j <= high) temp[k++] = arr[j++];
    
    // 临时数组拷贝回原数组
    for (int p = 0; p < temp.size(); ++p) {
        arr[low + p] = temp[p];
    }
}

// 归并排序递归函数
void mergeSortHelper(vector<int>& arr, int low, int high) {
    if (low >= high) return; // 子数组长度为1，终止递归
    int mid = low + (high - low) / 2; // 避免溢出
    mergeSortHelper(arr, low, mid); // 左子数组
    mergeSortHelper(arr, mid + 1, high); // 右子数组
    merge(arr, low, mid, high); // 合并
}

// 对外接口
void mergeSort(vector<int>& arr) {
    if (arr.empty()) return;
    mergeSortHelper(arr, 0, arr.size() - 1);
}
```
#### 重点
- 时间复杂度：平均/最坏/最好均为 $O(n\log n)$，性能稳定；
- 空间复杂度 $O(n)$（临时数组），**稳定排序**、非原地排序；
- 适合链表排序（无需额外空间）、外部排序（数据量超内存）；
- 合并操作是核心，需保证合并过程的稳定性（`arr[i] <= arr[j]` 而非 `<`）。

### 7. 堆排序
#### 思路
利用“最大堆”的特性（根节点是最大值）：先将数组构建为最大堆；然后依次将堆顶（最大值）与堆尾元素交换，缩小堆的范围并重新调整堆结构；直到堆的范围为1，数组有序。

#### 代码
```cpp
// 调整堆：以i为根的子树满足最大堆性质
void heapify(vector<int>& arr, int n, int i) {
    int largest = i; // 根节点
    int left = 2 * i + 1; // 左孩子
    int right = 2 * i + 2; // 右孩子
    
    // 找最大元素的索引
    if (left < n && arr[left] > arr[largest]) {
        largest = left;
    }
    if (right < n && arr[right] > arr[largest]) {
        largest = right;
    }
    
    // 若最大元素不是根节点，交换并递归调整
    if (largest != i) {
        swap(arr[i], arr[largest]);
        heapify(arr, n, largest);
    }
}

void heapSort(vector<int>& arr) {
    int n = arr.size();
    // 构建最大堆（从最后一个非叶子节点开始）
    for (int i = n / 2 - 1; i >= 0; --i) {
        heapify(arr, n, i);
    }
    
    // 依次取出堆顶元素
    for (int i = n - 1; i > 0; --i) {
        swap(arr[0], arr[i]); // 堆顶（最大值）与堆尾交换
        heapify(arr, i, 0); // 调整剩余元素为最大堆
    }
}
```
#### 重点
- 时间复杂度：平均/最坏/最好均为 $O(n\log n)$，性能稳定；
- 空间复杂度 $O(1)$，**不稳定排序**、原地排序；
- 适合大规模数据且要求原地排序的场景；
- 构建堆的时间复杂度为 $O(n)$，调整堆的时间复杂度为 $O(\log n)$。

### 8. 计数排序
#### 思路
适用于整数且取值范围小的场景：先统计每个数值出现的次数；然后根据次数计算每个数值的最终位置；最后反向填充数组，保证稳定性。

#### 代码
```cpp
void countingSort(vector<int>& arr) {
    if (arr.empty()) return;
    // 找最大值和最小值，确定计数范围
    int maxVal = *max_element(arr.begin(), arr.end());
    int minVal = *min_element(arr.begin(), arr.end());
    int range = maxVal - minVal + 1;
    
    vector<int> count(range, 0); // 计数数组
    vector<int> output(arr.size()); // 输出数组
    
    // 统计每个元素出现次数
    for (int num : arr) {
        count[num - minVal]++;
    }
    
    // 计算前缀和，确定元素最终位置
    for (int i = 1; i < count.size(); ++i) {
        count[i] += count[i - 1];
    }
    
    // 反向填充，保证稳定性
    for (int i = arr.size() - 1; i >= 0; --i) {
        output[count[arr[i] - minVal] - 1] = arr[i];
        count[arr[i] - minVal]--;
    }
    
    // 拷贝回原数组
    arr = move(output);
}
```
#### 重点
- 时间复杂度：平均/最坏/最好均为 $O(n+k)$（$k$ 为数值范围）；
- 空间复杂度 $O(n+k)$，**稳定排序**、非原地排序；
- 仅适用于整数且 $k$ 较小时（如成绩排序、年龄排序）；
- 反向填充是保证稳定性的关键，避免相同元素相对位置改变。

### 9. 桶排序
#### 思路
将数据分配到多个“桶”中，每个桶内单独排序（可用插入排序）；最后按桶的顺序合并所有桶的元素。核心是数据分布均匀，否则退化为 $O(n^2)$。

#### 代码
```cpp
void bucketSort(vector<int>& arr) {
    if (arr.empty()) return;
    // 找最大值和最小值
    int maxVal = *max_element(arr.begin(), arr.end());
    int minVal = *min_element(arr.begin(), arr.end());
    
    // 桶的数量（可根据数据分布调整）
    int bucketNum = 5;
    vector<vector<int>> buckets(bucketNum);
    
    // 计算每个元素所属的桶
    int range = (maxVal - minVal + 1) / bucketNum + 1; // 每个桶的数值范围
    for (int num : arr) {
        int idx = (num - minVal) / range;
        buckets[idx].push_back(num);
    }
    
    // 每个桶内排序，合并结果
    int idx = 0;
    for (auto& bucket : buckets) {
        insertionSort(bucket); // 复用插入排序
        for (int num : bucket) {
            arr[idx++] = num;
        }
    }
}

// 复用插入排序（桶内小规模排序）
void insertionSort(vector<int>& arr) {
    int n = arr.size();
    for (int i = 1; i < n; ++i) {
        int key = arr[i];
        int j = i - 1;
        while (j >= 0 && arr[j] > key) {
            arr[j + 1] = arr[j];
            --j;
        }
        arr[j + 1] = key;
    }
}
```
#### 重点
- 时间复杂度：平均/最好 $O(n)$，最坏（数据集中在一个桶）$O(n^2)$；
- 空间复杂度 $O(n+k)$（$k$ 为桶数），**稳定排序**、非原地排序；
- 适合数据分布均匀的场景（如均匀分布的浮点数、整数）；
- 桶的数量和范围需根据数据分布调整，是效率的关键。

### 10. 基数排序
#### 思路
按“位”排序（个位、十位、百位...），每一位用计数排序/桶排序实现；从最低位到最高位依次排序，最终得到有序数组。适用于整数或字符串（按字符位）。

#### 代码
```cpp
// 获取数字的第digit位（个位digit=1，十位digit=10...）
int getDigit(int num, int digit) {
    return (num / digit) % 10;
}

void radixSort(vector<int>& arr) {
    if (arr.empty()) return;
    // 找最大值，确定最大位数
    int maxVal = *max_element(arr.begin(), arr.end());
    int maxDigit = 1;
    while (maxVal / 10 > 0) {
        maxDigit *= 10;
        maxVal /= 10;
    }
    
    // 按每一位排序（计数排序实现）
    for (int digit = 1; digit <= maxDigit; digit *= 10) {
        vector<int> count(10, 0); // 0-9共10个桶
        vector<int> output(arr.size());
        
        // 统计当前位的数字出现次数
        for (int num : arr) {
            int d = getDigit(num, digit);
            count[d]++;
        }
        
        // 前缀和，确定位置
        for (int i = 1; i < 10; ++i) {
            count[i] += count[i - 1];
        }
        
        // 反向填充，保证稳定性
        for (int i = arr.size() - 1; i >= 0; --i) {
            int d = getDigit(arr[i], digit);
            output[count[d] - 1] = arr[i];
            count[d]--;
        }
        
        arr = move(output);
    }
}
```
#### 重点
- 时间复杂度：平均/最坏/最好均为 $O(n×k)$（$k$ 为最大位数）；
- 空间复杂度 $O(n+k)$，**稳定排序**、非原地排序；
- 适合整数、字符串（按字符ASCII码）排序；
- 必须从低位到高位排序，且每一位的排序需保证稳定性。

### 11. 模拟C++ std::sort（introsort）
#### 思路
std::sort 是混合排序算法（introsort）：核心是快速排序，当递归深度超过阈值（$2×\log_2n$）时切换为堆排序，小规模子数组改用插入排序；保证最坏时间复杂度 $O(n\log n)$。

#### 重点
- 时间复杂度：平均/最坏/最好均为 $O(n\log n)$，无最坏 $O(n^2)$ 情况；
- 空间复杂度 $O(\log n)$（递归栈），**不稳定排序**、原地排序；
- C++标准库默认实现，是通用场景的最优选择；
- 混合了快速排序（高效）、堆排序（避免最坏情况）、插入排序（小规模优化）。

### 1. 顺序查找 (Linear Search)
#### 思路
从数组/容器的第一个元素开始，逐个遍历并与目标值比较，找到则返回索引；遍历结束未找到则返回-1。无数据有序性要求，逻辑最简单。

#### 代码
```cpp
int linearSearch(const vector<int>& arr, int target) {
    // 逐个遍历元素
    for (int i = 0; i < arr.size(); ++i) {
        if (arr[i] == target) {
            return i; // 找到目标，返回索引
        }
    }
    return -1; // 未找到
}
```
#### 重点
- 时间复杂度：平均/最坏 $O(n)$，最好（目标在首位）$O(1)$；
- 空间复杂度 $O(1)$，原地查找，无额外空间；
- 数据无任何要求（有序/无序均可），适配链表、数组等任意线性结构；
- 缺点是效率低，仅适合小规模数据或单次查找场景。

### 2. 二分查找 (Binary Search)
#### 思路
仅适用于**有序数组**：取数组中间元素与目标值比较，若目标值更小则在左半区间继续查找，若更大则在右半区间查找；重复缩小查找范围，直到找到目标或区间为空。

#### 代码
```cpp
int binarySearch(const vector<int>& arr, int target) {
    int left = 0;
    int right = arr.size() - 1;
    
    while (left <= right) {
        // 计算中间索引（避免 left+right 溢出）
        int mid = left + (right - left) / 2;
        if (arr[mid] == target) {
            return mid; // 找到目标，返回索引
        } else if (arr[mid] < target) {
            left = mid + 1; // 目标在右半区间
        } else {
            right = mid - 1; // 目标在左半区间
        }
    }
    return -1; // 未找到
}
```
#### 重点
- 时间复杂度：平均/最坏 $O(\log n)$，最好（目标在中间）$O(1)$；
- 空间复杂度 $O(1)$（迭代版），递归版为 $O(\log n)$（递归栈）；
- 核心要求：数据**有序且支持随机访问**（数组），链表无法高效实现；
- 优势是查找效率极高，适合静态有序数组（查找频繁、修改少）。

### 3. 插值查找 (Interpolation Search)
#### 思路
二分查找的优化版，适用于**有序且分布均匀**的数组：不再固定取中间值，而是根据目标值的位置动态计算查找点（类似数轴插值），公式为：  
$$
pos = left + \frac{(target - arr[left]) \times (right - left)}{arr[right] - arr[left]}
$$
分布均匀时，查找点更接近目标，效率远高于二分查找；分布不均时退化为顺序查找。

#### 代码
```cpp
int interpolationSearch(const vector<int>& arr, int target) {
    int left = 0;
    int right = arr.size() - 1;
    
    // 需满足：数组有序 + 目标在[arr[left], arr[right]]范围内 + arr[left] != arr[right]
    while (left <= right && target >= arr[left] && target <= arr[right] && arr[left] != arr[right]) {
        // 插值计算查找位置
        int pos = left + ((target - arr[left]) * (right - left)) / (arr[right] - arr[left]);
        
        if (arr[pos] == target) {
            return pos; // 找到目标
        } else if (arr[pos] < target) {
            left = pos + 1; // 目标在右半区间
        } else {
            right = pos - 1; // 目标在左半区间
        }
    }
    
    // 处理边界情况（如所有元素相同）
    if (left <= right && arr[left] == target) {
        return left;
    }
    return -1; // 未找到
}
```
#### 重点
- 时间复杂度：平均 $O(\log \log n)$，最坏（分布极不均）$O(n)$，最好 $O(1)$；
- 空间复杂度 $O(1)$，原地查找；
- 核心要求：数组**有序且数值分布均匀**（如 1,2,3,4... 或 10,20,30...）；
- 适合大规模均匀分布的有序数组，比二分查找减少查找次数。

### 4. 斐波那契查找 (Fibonacci Search)
#### 思路
二分查找的变种，利用斐波那契数列分割区间：用斐波那契数 $F[k]$ 确定查找点（$mid = left + F[k-1] - 1$），避免二分查找的除法运算（仅用加减法），更适配硬件底层；同时内存访问局部性更好，缓存命中率更高。

#### 代码
```cpp
// 生成斐波那契数列，直到大于等于数组长度
void generateFibonacci(vector<int>& fib, int n) {
    fib.push_back(0);
    fib.push_back(1);
    while (fib.back() < n) {
        int next = fib[fib.size()-1] + fib[fib.size()-2];
        fib.push_back(next);
    }
}

int fibonacciSearch(const vector<int>& arr, int target) {
    int n = arr.size();
    if (n == 0) return -1;
    
    vector<int> fib;
    generateFibonacci(fib, n);
    
    int k = fib.size() - 1; // 斐波那契数列索引
    // 扩展数组长度到fib[k]-1（补最后一个元素）
    int offset = 0; // 偏移量
    
    while (fib[k] > 1) {
        int mid = offset + fib[k-1] - 1;
        // 避免mid越界
        mid = min(mid, n - 1);
        
        if (arr[mid] == target) {
            return mid; // 找到目标
        } else if (arr[mid] < target) {
            // 目标在右半区间，调整斐波那契索引和偏移
            k = k - 2;
            offset = mid + 1;
        } else {
            // 目标在左半区间，调整斐波那契索引
            k = k - 1;
        }
    }
    
    // 检查最后一个元素
    if (fib[1] && arr[offset] == target) {
        return offset;
    }
    return -1; // 未找到
}
```
#### 重点
- 时间复杂度：平均/最坏 $O(\log n)$，最好 $O(1)$；
- 空间复杂度 $O(1)$（可优化斐波那契数列生成，仅保留前两个值）；
- 核心优势：仅用加减法，无除法/乘法，硬件执行效率更高；内存访问局部性好；
- 适用场景：对运算效率（如嵌入式系统）、缓存命中率有要求的有序数组查找。

### 5. 哈希查找 (Hash Search)
#### 思路
核心是“空间换时间”：先构建哈希表（键=数组元素，值=元素索引），查找时直接通过哈希函数计算目标值的哈希地址，访问哈希表即可得到结果；处理哈希冲突采用链地址法（每个哈希桶存储链表）。

#### 重点
- 时间复杂度：平均 $O(1)$，最坏（所有元素哈希冲突）$O(n)$，最好 $O(1)$；
- 空间复杂度 $O(n)$（哈希表占用额外空间）；
- 核心优势：查找速度极快，是高频查找场景的最优选择；
- 注意事项：哈希函数设计（避免冲突）、冲突处理（链地址法/开放寻址法）；
- 适用场景：高频查找、允许额外空间、对速度要求极高的场景（如缓存、数据库索引）。

### 6. 分块查找 (Block Search)
#### 思路
结合顺序查找和二分查找的优点：将数组分为若干块，要求**块间有序、块内无序**；先构建“块索引表”（存储每块的最大值和块起始位置），查找时先二分/顺序查找索引表确定目标所在块，再在块内顺序查找。

#### 代码
```cpp
// 块索引结构体：存储块的最大值和起始索引
struct BlockIndex {
    int maxVal;   // 块内最大值
    int startIdx; // 块起始索引
};

int blockSearch(const vector<int>& arr, int target) {
    if (arr.empty()) return -1;
    
    // 1. 分块（示例：每块大小为4，可根据数据调整）
    int blockSize = 4;
    int blockNum = (arr.size() + blockSize - 1) / blockSize; // 块数（向上取整）
    vector<BlockIndex> indexTable; // 块索引表
    
    // 2. 构建块索引表
    for (int i = 0; i < blockNum; ++i) {
        int start = i * blockSize;
        int end = min((i + 1) * blockSize - 1, (int)arr.size() - 1);
        int maxVal = arr[start];
        // 找块内最大值
        for (int j = start; j <= end; ++j) {
            if (arr[j] > maxVal) {
                maxVal = arr[j];
            }
        }
        indexTable.push_back({maxVal, start});
    }
    
    // 3. 查找块索引表（顺序查找，也可改用二分）
    int blockIdx = -1;
    for (int i = 0; i < indexTable.size(); ++i) {
        if (target <= indexTable[i].maxVal) {
            blockIdx = i;
            break;
        }
    }
    if (blockIdx == -1) return -1; // 目标超出所有块最大值
    
    // 4. 块内顺序查找
    int start = indexTable[blockIdx].startIdx;
    int end = min((blockIdx + 1) * blockSize - 1, (int)arr.size() - 1);
    for (int i = start; i <= end; ++i) {
        if (arr[i] == target) {
            return i; // 找到目标
        }
    }
    return -1; // 未找到
}
```
#### 重点
- 时间复杂度：平均 $O(\sqrt{n})$，最坏 $O(n)$（目标在最后一块且块内全遍历），最好 $O(1)$；
- 空间复杂度 $O(1)$（仅存储块索引表，可忽略）；
- 核心要求：块间有序、块内无序，块大小需合理（通常取 $\sqrt{n}$）；
- 优势：支持动态插入删除（只需调整对应块），适合数据量较大且动态变化的场景。

### 7. 二叉搜索树查找 (BST Search)
#### 思路
二叉搜索树（BST）特性：左子树所有节点值 < 根节点值，右子树所有节点值 > 根节点值；查找时从根节点开始，目标值小于当前节点则递归/迭代查左子树，大于则查右子树，等于则找到；最坏情况（树退化为链表）效率极低。

#### 重点
- 时间复杂度：平均 $O(\log n)$，最坏（树退化）$O(n)$，最好 $O(1)$；
- 空间复杂度 $O(h)$（$h$ 为树高，迭代版 $O(1)$，递归版 $O(h)$）；
- 核心特性：支持动态插入/删除，天然适合动态数据集；
- 缺点：未平衡的BST可能退化，实际开发中优先用平衡树（如下文AVL/红黑树）。

### 8. 平衡树查找 (AVL树，红黑树简化版)
#### 思路
AVL树是**自平衡二叉搜索树**，要求任意节点的左右子树高度差不超过1；插入/删除时通过旋转（左旋/右旋）保持平衡，保证查找效率稳定在 $O(\log n)$；以下实现简化版AVL树查找（核心逻辑与红黑树一致，均为平衡BST）。

#### 重点
- 时间复杂度：平均/最坏 $O(\log n)$，最好 $O(1)$；
- 空间复杂度 $O(h)$（$h=\log n$，平衡树高度）；
- 核心优势：自平衡，避免BST退化，查找效率稳定；
- 应用场景：C++ `std::map`/`std::set` 底层（红黑树，弱平衡，插入/删除效率更高），动态数据集的高效查找。

### 9. B+树查找 (简化版，适配内存)
#### 思路
B+树是**多路平衡查找树**，所有数据都存储在叶子节点，且叶子节点通过链表相连；非叶子节点仅存储索引（关键字+子节点指针），查找时从根节点开始，按关键字找到对应子节点，直到叶子节点；适合磁盘IO场景（减少磁盘访问次数），是数据库索引、文件系统的核心。

#### 重点
- 时间复杂度：$O(\log_t n)$（$t$ 为B+树阶数，远大于2，高度远低于AVL树）；
- 空间复杂度 $O(n)$（存储索引和数据）；
- 核心优势：高度低（减少IO次数）、叶子节点链表支持范围查询；
- 应用场景：数据库索引（MySQL InnoDB）、文件系统（NTFS/Ext4），大规模数据的磁盘级查找。

### 10. 模拟C++ std::binary_search
#### 思路
C++标准库 `std::binary_search` 是二分查找的标准化实现，仅返回“是否存在目标值”（而非索引）；要求输入容器支持随机访问迭代器（如 `vector`/`array`），且元素已排序；底层实现为优化版二分查找，保证 $O(\log n)$ 复杂度。

#### 重点
- 时间复杂度：平均/最坏 $O(\log n)$，最好 $O(1)$；
- 空间复杂度 $O(1)$，无额外空间；
- 核心要求：容器支持随机访问、元素已排序；
- 优势：标准化、高性能，是C++中有序容器查找的首选；
- 注意：`std::binary_search` 仅返回bool，如需索引用 `lower_bound`/`upper_bound`。

### 随机选择算法
随机选择算法（也叫快速选择算法）是基于快速排序的分治思想，核心目标是**在未排序数组中找到第 k 小（或第 k 大）的元素**，无需对整个数组排序，平均时间复杂度远优于先排序再取值的方式。

#### 思路
1. **分治核心**：借鉴快速排序的分区（Partition）操作，随机选择一个基准值（pivot），将数组分为“小于基准”“等于基准”“大于基准”三部分；
2. **缩小范围**：
   - 若“小于基准”的元素数量 ≥ k，说明第 k 小元素在左半区，递归处理左半区；
   - 若“小于基准+等于基准”的元素数量 < k，说明第 k 小元素在右半区，递归处理右半区（k 需减去左半区+基准的数量）；
   - 否则，基准值就是第 k 小元素；
3. **随机基准优化**：随机选择基准值，避免最坏情况（如数组已排序时固定选第一个元素导致 $O(n^2)$ 复杂度）。

#### 代码
```cpp
// 随机分区函数：返回基准值的最终索引，将数组分为[<pivot], [=pivot], [>pivot]
int randomPartition(vector<int>& arr, int low, int high) {
    // 随机选择基准值（避免固定选第一个元素的最坏情况）
    srand(time(0)); // 初始化随机数种子
    int randIdx = low + rand() % (high - low + 1);
    swap(arr[randIdx], arr[low]); // 交换到区间首部，复用快速排序分区逻辑
    
    int pivot = arr[low]; // 基准值
    int i = low, j = high;
    while (i < j) {
        // 从右找小于基准的元素
        while (i < j && arr[j] >= pivot) j--;
        arr[i] = arr[j];
        // 从左找大于基准的元素
        while (i < j && arr[i] <= pivot) i++;
        arr[j] = arr[i];
    }
    arr[i] = pivot; // 基准归位
    return i;
}

// 随机选择核心函数：找arr[low..high]中第k小的元素（k从1开始）
int quickSelect(vector<int>& arr, int low, int high, int k) {
    if (low == high) return arr[low]; // 区间只有一个元素，直接返回
    
    int pivotIdx = randomPartition(arr, low, high); // 分区，得到基准索引
    int leftSize = pivotIdx - low + 1; // 左半区（≤基准）的元素数量
    
    if (k == leftSize) {
        // 第k小元素就是基准值
        return arr[pivotIdx];
    } else if (k < leftSize) {
        // 第k小元素在左半区，递归处理
        return quickSelect(arr, low, pivotIdx - 1, k);
    } else {
        // 第k小元素在右半区，递归处理（k需减去左半区数量）
        return quickSelect(arr, pivotIdx + 1, high, k - leftSize);
    }
}

// 对外接口：找数组中第k小的元素（k从1开始）
int randomSelect(vector<int>& arr, int k) {
    if (arr.empty() || k < 1 || k > arr.size()) {
        throw invalid_argument("k is out of range"); // 异常处理：k非法
    }
    return quickSelect(arr, 0, arr.size() - 1, k);
}
```

#### 重点
1. **核心特性**：
   - 无需完整排序，仅聚焦目标元素所在区间，是找“第 k 小/大元素”的最优算法；
   - 稳定性：算法本身不稳定（分区会交换元素位置），但不影响“找第 k 小元素”的核心目标；
   - 原地操作：分区过程仅交换元素，无需额外数组空间。
2. **适用场景**：
   - 快速找到数组的中位数（k = n/2）、TOP-K 问题（找前 k 小/大元素，可结合随机选择+局部排序）；
   - 大数据量下的Top-K筛选（如从1000万条数据中找第100小的元素，无需全排序）；
   - 工程优化：小规模区间（如元素数量 < 10）可直接用插入排序，进一步提升效率。
