# 图-遍历

## DFS

### 邻接表

```c++
void dfs(const vector<vector<pair<int, int>>& adj, int start) {
    vector<bool> vis(adj.size(), false);
    stack<int> st;
    st.push(start);
    vis[start] = true;
    while (!st.empty()) {
        int u = st.top(); st.pop();
        cout << u << " "; // 处理该节点
        for (auto& [v, w] : adj[u]) {
            if (!vis[v]) {
                vis[v] = true;
                // 处理该节点与相邻节点的关系
                st.push(v);
            }
        }
    }
}
```

### 邻接矩阵

```c++
void dfs_mat(const vector<vector<int>>& mat, int start) {
    vector<bool> vis(mat.size(), false);
    stack<int> st;
    st.push(start);
    vis[start] = true;
    while (!st.empty()) {
        int u = st.top(); st.pop();
        cout << u << " "; // 处理该节点
        for (int v = 0; v < mat.size(); ++v) {
            if (mat[u][v] != 0 && !vis[v]) {
                vis[v] = true;
                // 处理该节点与相邻节点的关系
                st.push(v);
            }
        }
    }
}
```

### 链式前向星

```c++
void dfs(const StarGraph& graph, int start) {
    vector<bool> vis(graph.head.size(), false);
    stack<int> st;
    st.push(start);
    vis[start] = true;
    while (!st.empty()) {
        int u = st.top(); st.pop();
        cout << u << " ";
        for (int i = graph.head[u]; ~i; i = graph.edges[i].next) { // ~i等价于 i != -1
            int v = graph.edges[i].to;
            if (!vis[v]) {
                vis[v] = true;
                st.push(v);
            }
        }
    }
}
```
