## 创建图

### 邻接表

```c++
vector<vector<pair<int, int>>> graph(n + 1); // n为节点数
for (int i = 0; i < n - 1; ++i) {
	int u, v; // 节点
	int w; // w为权重
	cin >> u >> v >> w;
	graph[u].emplace_back(v, w); // u -> v
	graph[v].emplace_back(u, w); // v -> u
}
```

### 邻接矩阵

```c++
vector<vector<int>> graph(n + 1, vector<int>(n + 1, INT_MAX));
for (int i = 0; i < n - 1; ++i) {
    int u, v, w;
    cin >> u >> v >> w;
    graph[u][v] = w;
    graph[v][u] = w;
}
```

### 链式前向星

链式前向星本质是：**用数组模拟链表实现的邻接表**

```c++
constexpr int MAX_EDGE = 1000010;
constexpr int MAX_NODE = 1000010;
struct Edge {
    int to = 0;
    int weight = 0;
    int next = -1;
};
vector<Edge> edge(MAX_EDGE); // 边池
vector<int> head(MAX_NODE, -1); // 每个点的第一条边编号
int cnt = 0; // 边的计数器

void add(int u, int v, int w) noexcept {
    edge[++cnt] = { v, w, head[u] };
    head[u] = cnt;
}

// 建图
void build(int n) {
    cnt = 0;
    for (int i = 0; i < n - 1; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        add(u, v, w);
        add(v, u, w);
    }
}
```

### 边集数组

```c++
// C++17 构造函数 + 默认初始化 + 聚合初始化
struct Edge {
    int u = 0;
    int v = 0;
    int w = 0;
    Edge(int u_, int v_, int w_) : u(u_), v(v_), w(w_) {}
    // 按权值从小到大排序
    bool operator<(const Edge& other) const noexcept {
        return w < other.w;
    }
};
vector<Edge> edges;
// 建图（n 个点，n-1 条边，树）
void build(int n) {
    edges.reserve(n);
    for (int i = 0; i < n - 1; ++i) {
        int u, v, w;
        cin >> u >> v >> w;
        edges.emplace_back(u, v, w);
    }
}
sort(edges.begin(), edges.end()); // 排序（Kruskal 必备）
```

## DFS

### 邻接表

```c++
void dfs(const vector<vector<pair<int, int>>& adj, int start) {
    vector<bool> vis(adj.size(), false);
    stack<int> st;
    st.push(start);
    vis[start] = true;
    while (!st.empty()) {
        int u = st.top(); st.pop();
        cout << u << " "; // 处理该节点
        for (auto& [v, w] : adj[u]) {
            if (!vis[v]) {
                vis[v] = true;
                // 处理该节点与相邻节点的关系
                st.push(v);
            }
        }
    }
}
```

### 链式前向星

```c++
void dfs(const StarGraph& graph, int start) {
    vector<bool> vis(graph.head.size(), false);
    stack<int> st;
    st.push(start);
    vis[start] = true;
    while (!st.empty()) {
        int u = st.top(); st.pop();
        cout << u << " ";
        for (int i = graph.head[u]; ~i; i = graph.edges[i].next) { // ~i等价于 i != -1
            int v = graph.edges[i].to;
            if (!vis[v]) {
                vis[v] = true;
                st.push(v);
            }
        }
    }
}
```
